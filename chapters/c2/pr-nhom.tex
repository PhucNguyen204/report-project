\subsubsection{Phân rã use case “Quản lý nhóm gia đình”}\label{sssec:pr-nhom}

Gói Quản lý nhóm gia đình đáp ứng yêu cầu chia sẻ danh sách và phân vai trong gia đình. Hình~\ref{fig:uc-nhom} thể hiện ba use case: \uc{UC05} Tạo nhóm và xem thành viên, \uc{UC06} Mời và tham gia nhóm bằng mã mời (tầng 1) và \uc{UC07} Rời nhóm, xóa thành viên và chuyển quyền. \uc{UC06} gộp hai phía của cùng một dữ liệu là mã mời (\br{BR06}): trưởng nhóm tạo, chia sẻ hoặc thu hồi mã; người dùng nhập, quét hoặc mở liên kết để gia nhập. \uc{UC07} chỉ cài đặt hai trường hợp rời nhóm là thành viên thường và trưởng nhóm duy nhất (giải tán nhóm); trưởng nhóm còn thành viên khác nhận \msg{MSG24} và phải chuyển quyền trước, theo \br{BR05} và \br{BR07}. Xóa thành viên và chuyển quyền cài đặt tối giản.

\diagram[0.85\textwidth]{c2-05-uc-nhom}{Biểu đồ use case phân rã gói Quản lý nhóm gia đình}{fig:uc-nhom}

\ucgoal{UC05}{Tạo nhóm và xem thành viên}{2}{Người dùng, Thành viên nhóm}{tạo nhóm, xem thông tin nhóm và thành viên}

\paragraph*{Tạo nhóm (\uc{UC05})}
Người dùng chưa thuộc nhóm gia đình nào có thể tạo nhóm với tên hợp lệ và trở thành trưởng nhóm; dữ liệu cá nhân chỉ được sao chép sang nhóm khi người dùng lựa chọn (\br{BR04}, \br{BR05}).

\ucgoal{UC06}{Mời và tham gia nhóm bằng mã mời}{1}{Trưởng nhóm, Người dùng}{mời thành viên, tham gia nhóm}
Hai biến thể của cùng một use case: A là phía trưởng nhóm tạo mã, B là phía người dùng nhập mã.
Hai use case thuộc hai tác nhân khác nhau nhưng chia sẻ cùng một dữ liệu là mã mời (\br{BR06}): trưởng nhóm tạo, chia sẻ hoặc thu hồi mã; người dùng nhập, quét hoặc mở liên kết để gia nhập.

\begin{ucspec}{UC06}{Mời và tham gia nhóm bằng mã mời}
\ucfield{Tác nhân}{Trưởng nhóm; Người dùng}
\ucfield{Mô tả}{\textbf{A.} Cho phép trưởng nhóm tạo mã mời 8 ký tự kèm mã QR và liên kết mời để người khác gia nhập nhóm; xem, chia sẻ, tạo lại hoặc thu hồi mã đang hiệu lực. \textbf{B.} Cho phép người dùng gia nhập một nhóm gia đình bằng mã mời nhập tay hoặc bằng cách quét mã QR.}
\ucfield{Điều kiện trước}{\textbf{A.} Người dùng đã đăng nhập và là trưởng nhóm của một nhóm gia đình (không phải không gian cá nhân). \textbf{B.} Người dùng đã đăng nhập và đã nhận được mã mời từ trưởng nhóm.}
\ucflow{Biến thể A – Mời thành viên vào nhóm: luồng thực thi chính}
\ucstep{1}{Trưởng nhóm}{Mở màn hình Gia đình, chọn “Mời thành viên”.}
\ucstep{2}{Hệ thống}{Kiểm tra người dùng là trưởng nhóm và nhóm chưa đủ số thành viên tối đa (\br{BR05}, \br{BR23}).}
\ucstep{3}{Hệ thống}{Nếu nhóm chưa có mã còn hiệu lực, sinh mã gồm 8 ký tự chữ in hoa và chữ số, loại các ký tự dễ nhầm, hiệu lực 48 giờ (\br{BR06}); hiển thị mã, mã QR, thời gian còn lại và các nút “Chia sẻ”, “Tạo mã mới”, “Thu hồi”.}
\ucstep{4}{Trưởng nhóm}{Chọn “Chia sẻ”.}
\ucstep{5}{Hệ thống}{Mở bảng chia sẻ của hệ điều hành với nội dung gồm mã mời và liên kết mời để \uc{UC06} nhận lại.}
\ucflow{Biến thể A – Mời thành viên vào nhóm: luồng thực thi mở rộng}
\ucstep{2a}{Hệ thống}{Người thực hiện không còn là trưởng nhóm (vừa chuyển quyền trên thiết bị khác): hiển thị \msg{MSG13}; kết thúc use case.}
\ucstep{2b}{Hệ thống}{Nhóm đã đủ số thành viên tối đa: hiển thị \msg{MSG09}; kết thúc use case.}
\ucstep{3a}{Hệ thống}{Nhóm đã có mã còn hiệu lực: hiển thị lại mã hiện có thay vì sinh mã mới.}
\ucstep{3b}{Hệ thống}{Mã hiện có đã hết hạn: hiển thị trạng thái hết hạn và chỉ sinh mã mới khi trưởng nhóm chọn “Tạo mã mới”.}
\ucstep{4a}{Trưởng nhóm}{Chọn “Tạo mã mới”: hiển thị \msg{MSG19} nêu rõ mã cũ sẽ bị vô hiệu; khi đồng ý, sinh mã mới và vô hiệu mã cũ (\br{BR06}).}
\ucstep{4b}{Trưởng nhóm}{Chọn “Thu hồi”: hiển thị \msg{MSG19}; khi đồng ý, mã bị vô hiệu ngay và người nhập mã sau đó nhận \msg{MSG07} ở \uc{UC06}.}
\ucstep{5a}{Trưởng nhóm}{Đóng bảng chia sẻ mà không gửi: mã vẫn còn hiệu lực và không có thay đổi dữ liệu.}
\ucflow{Biến thể B – Tham gia nhóm gia đình: luồng thực thi chính}
\ucstep{B1}{Người dùng}{Chọn “Tham gia nhóm” tại màn hình Gia đình.}
\ucstep{B2}{Hệ thống}{Hiển thị ô nhập mã mời và nút “Quét mã QR”.}
\ucstep{B3}{Người dùng}{Nhập mã mời gồm 8 ký tự và nhấn “Tiếp tục”.}
\ucstep{B4}{Hệ thống}{Tra cứu mã mời, kiểm tra mã tồn tại, còn hiệu lực và chưa bị thu hồi (\br{BR06}).}
\ucstep{B5}{Hệ thống}{Kiểm tra người dùng chưa thuộc nhóm gia đình nào và nhóm chưa đủ số thành viên tối đa (\br{BR04}, \br{BR05}).}
\ucstep{B6}{Hệ thống}{Hiển thị tên nhóm, tên trưởng nhóm, số thành viên hiện tại để người dùng xác nhận.}
\ucstep{B7}{Người dùng}{Nhấn “Tham gia”.}
\ucstep{B8}{Hệ thống}{Thêm người dùng vào nhóm với vai trò thành viên, chuyển không gian làm việc sang nhóm, gửi thông báo tới trưởng nhóm và hiển thị \msg{MSG10}.}
\ucflow{Biến thể B – Tham gia nhóm gia đình: luồng thực thi mở rộng}
\ucstep{B3a}{Người dùng}{Chọn “Quét mã QR”: hệ thống xin quyền truy cập camera, đọc mã từ QR và chuyển sang bước B4.}
\ucstep{B3b}{Người dùng}{Mở liên kết mời từ tin nhắn: hệ điều hành mở ứng dụng với mã đã điền sẵn và chuyển sang bước B4.}
\ucstep{B4a}{Hệ thống}{Mã không tồn tại, hết hạn hoặc đã bị thu hồi: hiển thị \msg{MSG07}; quay lại bước B3.}
\ucstep{B5a}{Hệ thống}{Người dùng đang thuộc một nhóm gia đình khác: hiển thị \msg{MSG08}; kết thúc use case.}
\ucstep{B5b}{Hệ thống}{Nhóm đã đủ số thành viên: hiển thị \msg{MSG09}; kết thúc use case.}
\ucstep{B7a}{Người dùng}{Nhấn “Hủy”: kết thúc use case, không có thay đổi.}
\ucfield{Điều kiện sau}{\textbf{A.} Nhóm có đúng một mã còn hiệu lực, hoặc không còn mã nào sau khi thu hồi; số thành viên không thay đổi cho đến khi có người gia nhập qua \uc{UC06}. \textbf{B.} Người dùng là thành viên của nhóm, có quyền đọc và ghi dữ liệu chung của nhóm; dữ liệu trong không gian cá nhân được giữ nguyên.}
\ucfield{Quy tắc nghiệp vụ}{\br{BR04}, \br{BR05}, \br{BR06}, \br{BR23}}
\end{ucspec}



\diagram[0.7\textwidth]{c2-06-act-tham-gia-nhom}{Biểu đồ hoạt động của use case Tham gia nhóm gia đình}{fig:act-tham-gia-nhom}

Biểu đồ hoạt động ở Hình~\ref{fig:act-tham-gia-nhom} thể hiện thứ tự kiểm tra ba điều kiện: mã còn hiệu lực, người dùng chưa là thành viên của nhóm khác và nhóm còn chỗ. Hệ thống kiểm tra các điều kiện này trước khi hiển thị bước xác nhận, tránh yêu cầu người dùng tiếp tục khi thao tác sẽ bị từ chối. Các chức năng \uc{UC05} đến \uc{UC07} được đặc tả tại Phụ lục~\ref{pl:dac-ta-bo-sung}; \uc{UC07} chỉ được cài đặt ở mức tối giản.
